cmake_minimum_required(VERSION 3.20)
project(voice_participant_tap LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(NODE_INCLUDE_DIR "" CACHE PATH "Folder containing node_api.h from Node 24 headers")
if(NOT NODE_INCLUDE_DIR)
  message(FATAL_ERROR "Pass -DNODE_INCLUDE_DIR=<Node include folder>")
endif()
add_subdirectory(minhook)
add_library(participant_tap SHARED participant_tap.cpp)
target_include_directories(participant_tap PRIVATE "${NODE_INCLUDE_DIR}")
target_link_libraries(participant_tap PRIVATE minhook advapi32)
target_compile_definitions(participant_tap PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(participant_tap PROPERTIES PREFIX "" SUFFIX ".node")
add_executable(callback_test callback_test.cpp)
target_compile_options(callback_test PRIVATE /UNDEBUG)
